    <script>
        let fbDb = null;
        let map, currentLayer, drawingLayers=[], activeType=null;
        let perimeterCenter=[37.5407,-77.4360], perimeterCircle=null, centerRedDot=null;
        let userMarker=null, watchId=null;
        let currentUserName='UNKNOWN', currentUserDept='N/A';
        let currentUserId = 'user-' + Math.random().toString(36).substr(2,9);
        let otherMarkers={};

        const firebaseConfig = {
            apiKey: "AIzaSyDFzxwNI8eLhjlsZQf_sKpboe_zqLkNZfg",
            authDomain: "tacnet-law-enforcement.firebaseapp.com",
            databaseURL: "https://tacnet-law-enforcement-default-rtdb.firebaseio.com",
            projectId: "tacnet-law-enforcement",
            storageBucket: "tacnet-law-enforcement.firebasestorage.app",
            messagingSenderId: "85046927848",
            appId: "1:85046927848:web:8ebe08d72b9d252e50445d",
            measurementId: "G-KMZK35RRCY"
        };

        try {
            firebase.initializeApp(firebaseConfig);
            fbDb = firebase.database();
            document.getElementById('firebaseStatus').innerText = "FIREBASE: CONNECTED - ID " + currentUserId;
            console.log("Firebase OK", currentUserId);
        } catch(e) {
            console.error("Firebase init error", e);
            document.getElementById('firebaseStatus').innerText = "FIREBASE ERROR: " + e.message;
        }

        function enterApp(){
            document.getElementById('landingPage').classList.add('hidden');
            document.getElementById('loginPage').classList.remove('hidden');
        }
        function toggleAccessCode(){
            let role=document.getElementById('roleSelect').value;
            let codeInput=document.getElementById('accessCode');
            if(role==='admin' || role==='commander') codeInput.classList.remove('hidden');
            else codeInput.classList.add('hidden');
        }
        function attemptLogin(){
            let role=document.getElementById('roleSelect').value;
            let code=document.getElementById('accessCode').value;
            if(role==='admin' && code!=='1041'){alert("INVALID ADMIN CODE");return;}
            if(role==='commander' && code!=='1042'){alert("INVALID COMMANDER CODE");return;}
            currentUserName=document.getElementById('loginName').value || 'OFFICER-' + currentUserId.substr(-4);
            currentUserDept=document.getElementById('loginDept').value || 'N/A';
            startApp();
        }
        function startApp(){
            document.getElementById('loginPage').classList.add('hidden');
            document.getElementById('appContainer').classList.remove('hidden');
            setTimeout(initMap,100);
        }
        function exitApp(){ if(fbDb && watchId!==null){ fbDb.ref('officers/'+currentUserId).remove(); } location.reload(); }
        function showToast(msg){ let t=document.getElementById('toast'); t.innerText=msg; t.style.display='block'; clearTimeout(t.timer); t.timer=setTimeout(()=>t.style.display='none',3000); }

        function initMap(){
            map = L.map('map',{zoomControl:false,attributionControl:false,maxZoom:22}).setView(perimeterCenter,13);
            L.control.zoom({position:'topright'}).addTo(map);
            currentLayer = L.tileLayer('https://server.arcgisonline.com/ArcGIS/rest/services/World_Imagery/MapServer/tile/{z}/{y}/{x}',{maxZoom:22}).addTo(map);
            drawPerimeter(2500);
            
            // FIXED REAL-TIME LISTENER - This was broken before
            if(fbDb){
                showToast("MULTIPLAYER TOWER: CONNECTED");
                fbDb.ref('officers').on('value',(snapshot)=>{
                    const data=snapshot.val() || {};
                    document.getElementById('liveCount').innerText = Object.keys(data).length + " LIVE";
                    
                    // Remove old markers not in DB
                    for(let id in otherMarkers){
                        if(!data[id]){ map.removeLayer(otherMarkers[id]); delete otherMarkers[id]; }
                    }
                    // Add / update markers
                    for(let id in data){
                        if(id===currentUserId) continue; // Don't show self twice
                        let off=data[id];
                        if(off && off.lat && off.lng){
                            if(otherMarkers[id]){
                                otherMarkers[id].setLatLng([off.lat, off.lng]);
                                otherMarkers[id].getTooltip().setContent(off.name + " (" + off.dept + ")");
                            } else {
                                // GREEN for other devices, BLUE for you - so you can tell them apart
                                let mIcon = L.divIcon({
                                    className:'gps-flash-icon',
                                    html:'<div class="gps-pulse-ring" style="background:rgba(0,255,0,0.4);"></div><div class="gps-dot-center" style="background:#00ff41;box-shadow:0 0 8px #00ff41;"></div>',
                                    iconSize:[20,20], iconAnchor:[10,10]
                                });
                                otherMarkers[id] = L.marker([off.lat, off.lng],{icon:mIcon}).addTo(map);
                                otherMarkers[id].bindTooltip(off.name, {permanent:true,direction:'top',offset:[0,-10]});
                                showToast("NEW OFFICER JOINED: " + off.name);
                            }
                        }
                    }
                });
                // Auto-remove on disconnect
                fbDb.ref('.info/connected').on('value',(snap)=>{
                    if(snap.val()===true){
                        fbDb.ref('officers/'+currentUserId).onDisconnect().remove();
                    }
                });
            } else {
                showToast("MULTIPLAYER TOWER: OFFLINE - CHECK FIREBASE");
            }

            map.on('click',e=>{
                if(activeType){
                    let m = L.circleMarker(e.latlng,{radius:12,fillColor:'blue',color:'white',fillOpacity:1,weight:2}).addTo(map);
                    m.bindTooltip(activeType,{permanent:true,direction:'top'});
                    m.on('click',()=>map.removeLayer(m));
                    activeType=null;
                }
            });
        }

        function searchAddr(){
            let q=document.getElementById('addrInput').value;
            fetch(`https://nominatim.openstreetmap.org/search?format=json&q=${encodeURIComponent(q)}`)
                .then(r=>r.json()).then(d=>{ if(d[0]){ perimeterCenter=[d[0].lat,d[0].lon]; map.setView(perimeterCenter,15); drawPerimeter(document.getElementById('radiusSlider').value); showToast("TARGET LOCATED"); } });
        }
        function drawPerimeter(ft){
            if(perimeterCircle) map.removeLayer(perimeterCircle);
            if(centerRedDot) map.removeLayer(centerRedDot);
            perimeterCircle = L.circle(perimeterCenter,{radius:ft/3.28084,color:'#00ff41',fillColor:'#00ff41',fillOpacity:0.1,weight:3,dashArray:'5,5',interactive:false}).addTo(map);
            centerRedDot = L.circleMarker(perimeterCenter,{radius:6,fillColor:'red',color:'red',fillOpacity:1,weight:1,interactive:false}).addTo(map);
        }
        function adjustPerimeter(val){ document.getElementById('radiusVal').innerText=val+" FT"; drawPerimeter(parseFloat(val)); }
        function switchLayer(type){
            map.removeLayer(currentLayer);
            document.querySelectorAll('.top-toggles .btn-toggle').forEach(b=>b.classList.remove('active'));
            if(type==='str'){ currentLayer=L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png',{maxZoom:22}).addTo(map); document.getElementById('t-str').classList.add('active'); }
            if(type==='sat'){ currentLayer=L.tileLayer('https://server.arcgisonline.com/ArcGIS/rest/services/World_Imagery/MapServer/tile/{z}/{y}/{x}',{maxZoom:22}).addTo(map); document.getElementById('t-sat').classList.add('active'); }
            if(type==='ter'){ currentLayer=L.tileLayer('https://basemap.nationalmap.gov/arcgis/rest/services/USGSTopo/MapServer/tile/{z}/{y}/{x}',{maxZoom:22}).addTo(map); document.getElementById('t-ter').classList.add('active'); }
        }
        function toggleNVG(){ document.getElementById('map').classList.toggle('nvg-active'); document.getElementById('t-nvg').classList.toggle('active'); }
        function openMarkerModal(){ document.getElementById('markerModal').classList.remove('hidden'); }
        function placeMarker(type){ activeType=type; document.getElementById('markerModal').classList.add('hidden'); showToast("TAP MAP TO DROP "+type); }
        function clearMap(){ map.eachLayer(l=>{ if(l instanceof L.CircleMarker && l!==centerRedDot) map.removeLayer(l); }); }

        function toggleLiveGPS(){
            let btn=document.getElementById('btn-gps');
            if(watchId!==null){
                navigator.geolocation.clearWatch(watchId); watchId=null; btn.classList.remove('active'); showToast("LIVE GPS STOPPED");
                if(fbDb) fbDb.ref('officers/'+currentUserId).remove();
                if(userMarker){ map.removeLayer(userMarker); userMarker=null; }
                return;
            }
            if(!navigator.geolocation){ showToast("GPS NOT SUPPORTED"); return; }
            btn.classList.add('active');
            showToast("REQUESTING GPS... TAP ALLOW");
            watchId=navigator.geolocation.watchPosition(pos=>{
                let latlng=[pos.coords.latitude,pos.coords.longitude];
                if(!userMarker){
                    let flashIcon = L.divIcon({className:'gps-flash-icon',html:'<div class="gps-pulse-ring"></div><div class="gps-dot-center" style="background:#00ffff;box-shadow:0 0 8px #00ffff;"></div>',iconSize:[20,20],iconAnchor:[10,10]});
                    userMarker=L.marker(latlng,{icon:flashIcon}).addTo(map);
                    userMarker.bindTooltip("YOU - "+currentUserName,{permanent:true,direction:'top',offset:[0,-10]});
                    map.setView(latlng,18);
                    showToast("GPS LOCK ACQUIRED - BROADCASTING TO TEAM");
                } else { userMarker.setLatLng(latlng); }
                if(fbDb){
                    fbDb.ref('officers/'+currentUserId).set({name:currentUserName,dept:currentUserDept,lat:pos.coords.latitude,lng:pos.coords.longitude,timestamp:Date.now()});
                }
            },err=>{
                if(err.code===1){ showToast("GPS BLOCKED: Check iPhone Settings > Privacy > Location"); btn.classList.remove('active'); if(watchId) navigator.geolocation.clearWatch(watchId); watchId=null; }
            },{enableHighAccuracy:true,maximumAge:0,timeout:60000});
        }
        function triggerOfficerDown(){
            document.getElementById('downName').innerText=currentUserName;
            document.getElementById('downDept').innerText="DEPT: "+currentUserDept;
            if(userMarker){ let ll=userMarker.getLatLng(); document.getElementById('downGps').innerText="GPS: "+ll.lat.toFixed(5)+", "+ll.lng.toFixed(5); }
            document.getElementById('downAlert').style.display='flex';
        }
        function cancelOfficerDown(){ document.getElementById('downAlert').style.display='none'; }
        window.addEventListener('beforeunload',()=>{ if(fbDb && watchId!==null){ fbDb.ref('officers/'+currentUserId).remove(); } });
